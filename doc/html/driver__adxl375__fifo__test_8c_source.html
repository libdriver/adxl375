74 break;

83 adxl375_interface_debug_print("adxl375: read failed.\n");

85 return;
86 }
87 else
88 {
89 gs_watermark_flag++;
90 if (gs_watermark_flag > 2)

92 if (adxl375_set_measure(&gs_handle, ADXL375_BOOL_FALSE) != 0)
93 {
94 adxl375_interface_debug_print("adxl375: set measure failed.\n");
95
96 return;
97 }
98 }
99 }
100 adxl375_interface_debug_print("adxl375: irq water mark with %d.\n", len);
101
102 break;
103 }
104 case ADXL375_INTERRUPT_OVERRUN :
105 {
106 break;
107 }
108 default :
109 {
110 break;
111 }
112 }
